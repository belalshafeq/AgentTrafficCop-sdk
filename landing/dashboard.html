Quick Start

Integrate AgentTrafficCop in one line of code.

from agenttrafficcop import Guard, LoopDetected

guard = Guard(
    url="https://api.agenttrafficcop.com",
    api_key="YOUR_API_KEY_HERE"
)

try:
    await guard.check(
        session_id="test-run",
        sender="coder",
        recipient="reviewer",
        payload={"diff": "fix typo"}
    )
    print("Approved!")
except LoopDetected as e:
    print(f"Loop caught: {e.error_code}")
